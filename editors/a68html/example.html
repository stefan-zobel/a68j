1{ example.a68 - shows the highlighting of GNU Algol 68 in SUPPER stropping.
 2  Comments are in curly braces { and can be nested };
 3  a "}" in quotes does not end the comment.
 4  Compile: ga68 -fstropping=supper example.a68 -o example }
 5begin
 6   mode Node = struct (int value, ref Node next);
 7   mode Shape = union (Circle, Square);
 8   mode Circle = struct (real radius);
 9   mode Square = struct (real side);
10
11   prio MAX = 9;
12   op MAX = (int a, b) int: (a > b | a | b);
13   op +:= = (ref ref Node list, int v) ref Node:
14      list := heap Node := (v, list);
15
16   proc area = (Shape s) real:
17      case s
18      in (Circle c): pi * radius of c ** 2,
19         (Square q): side of q * side of q
20      esac;
21
22   proc (int) bool is_prime = (int n) bool:
23      if n < 2 then false
24      elif n = 2 then true
25      else
26         bool prime := ODD n;
27         for d from 3 by 2 while prime andth d * d <= n
28         do prime := n MOD d /= 0 od;
29         prime
30      fi;
31
32   ref Node list := nil;
33   for i to 20 do if is_prime (i) then list +:= i fi od;
34
35   int count := 0;
36   ref Node p := list;
37   while ref Node (p) isnt nil
38   do count +:= 1;
39      puts (whole (value of p, 0) + " ");
40      p := next of p
41   od;
42   puts ("'n");
43
44   long int big = long_max_int;
45   bits mask = 16rff00 AND 2r1111 0000 0000;
46   real r = 1.5e-3 + .25 + 1 000;
47   Circle unit; radius of unit := 1.0;
48   puts ("max: " + whole (3 MAX 7, 0) + ", area: "
49         + fixed (area (unit), 0, 4) + "'n");
50   puts ("""quoted"" and '(u00e4) with a tab:'t|'n");
51   puts (whole (ABS mask, 0) + " " + fixed (r, 0, 4) + " "
52         + whole (big MOD LENG 1000, 0) + "'n");
53   if count > 100 then goto done fi;
54   puts ("finished'n");
55done:
56   skip
57end
